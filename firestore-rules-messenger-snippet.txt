// ВСТАВИТЬ в firestore.rules внутрь match /databases/{database}/documents { ... }.
// Мессенджерное ПИШЕТ только сервер (Admin SDK обходит правила), клиент только читает свои данные.

match /messengerAccounts/{uid} {
  allow read: if request.auth != null;
  allow write: if false;
}
match /messengerHandles/{handle} {
  allow read, write: if false;
}
match /messengerChats/{chatId} {
  allow read: if request.auth != null && request.auth.uid in resource.data.members;
  allow write: if false;
  match /messages/{msgId} {
    allow read: if request.auth != null && request.auth.uid in get(/databases/$(database)/documents/messengerChats/$(chatId)).data.members;
    allow write: if false;
  }
  match /typing/{uid} {
    allow read: if request.auth != null && request.auth.uid in get(/databases/$(database)/documents/messengerChats/$(chatId)).data.members;
    allow write: if false;
  }
}
match /messengerUserChats/{uid}/chats/{chatId} {
  allow read: if request.auth != null && request.auth.uid == uid;
  allow write: if false;
}
match /messengerUserFolders/{uid}/folders/{folderId} {
  allow read: if request.auth != null && request.auth.uid == uid;
  allow write: if false;
}

// Этап 2–3: истории читаются ТОЛЬКО через API (нужна фильтрация по контактам), остальное только сервер.
match /messengerStories/{id}   { allow read, write: if false; }
match /messengerReports/{id}   { allow read, write: if false; }
match /messengerBlocks/{uid}/blocked/{other} { allow read, write: if false; }

// Каналы, Mili Premium, стикеры: всё через API.
match /messengerInvites/{code}          { allow read, write: if false; }
match /messengerStickerPacks/{slug}     { allow read, write: if false; }
match /messengerUserStickers/{uid}      { allow read, write: if false; }
// views подколлекция сообщений канала (messengerChats/{id}/messages/{msgId}/views) закрыта общим запретом: отдельный match не нужен.

// Опросы: свой голос читает только сам проголосовавший; ответ викторины (secret) и slow/views закрыты по умолчанию.
match /messengerChats/{chatId}/messages/{msgId}/votes/{uid} {
  allow read: if request.auth != null && request.auth.uid == uid
    && request.auth.uid in get(/databases/$(database)/documents/messengerChats/$(chatId)).data.members;
  allow write: if false;
}
match /messengerJoinRequests/{id} { allow read, write: if false; }
