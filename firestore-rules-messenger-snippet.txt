// ВСТАВИТЬ в firestore.rules внутрь match /databases/{database}/documents { ... }.
// Мессенджерное ПИШЕТ только сервер (Admin SDK обходит правила), клиент только читает свои данные.

match /messengerAccounts/{uid} {
  allow read: if request.auth != null;
  allow write: if false;
}
match /messengerHandles/{handle} {
  allow read, write: if false;
}
match /messengerChats/{chatId} {
  allow read: if request.auth != null && request.auth.uid in resource.data.members;
  allow write: if false;
  match /messages/{msgId} {
    allow read: if request.auth != null && request.auth.uid in get(/databases/$(database)/documents/messengerChats/$(chatId)).data.members;
    allow write: if false;
  }
  match /typing/{uid} {
    allow read: if request.auth != null && request.auth.uid in get(/databases/$(database)/documents/messengerChats/$(chatId)).data.members;
    allow write: if false;
  }
}
match /messengerUserChats/{uid}/chats/{chatId} {
  allow read: if request.auth != null && request.auth.uid == uid;
  allow write: if false;
}
match /messengerUserFolders/{uid}/folders/{folderId} {
  allow read: if request.auth != null && request.auth.uid == uid;
  allow write: if false;
}
